# --- Stock brokers (Singapore) ---

# `--check-broker` failed with ModuleNotFoundError, which reads as "SDK missing"

# ever run. `longport` is the current name; `longbridge` is the superseded one.

moomoo-api==10.9.6908   # moomoo OpenAPI — needs the moomoo OpenD gateway running locally
